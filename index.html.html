Step-by-Step IntegrationSupabase
1
Free • 2 min
Create a Supabase project
Go to supabase.com → New project. Copy your Project URL and anon public key from Settings → API.
2
In your HTML file
Add the Supabase SDK
Add this script tag in your <head> before your own script:
<script src="https://cdn.jsdelivr.net/npm/@supabase/supabase-js@2"></script>
3
Replace in your script
Initialize Supabase client
At the top of your <script> block, add:
// Replace these with your real values from supabase.com const SUPABASE_URL = 'https://xxxx.supabase.co' const SUPABASE_KEY = 'your-anon-public-key-here' const supabase = window.supabase.createClient(SUPABASE_URL, SUPABASE_KEY)
4
Replace doSignup()
Replace signup with Supabase Auth
async function doSignup() { // ... keep your validation code ... const { data, error } = await supabase.auth.signUp({ email: email, password: pass, options: { data: { first, last } } }) if (error) { showErr('su', error.message); return } enterDash({ first, last, email, initials: (first[0]+last[0]).toUpperCase() }) }
5
Replace doLogin()
Replace login with Supabase Auth
async function doLogin() { const email = document.getElementById('li-email').value.trim() const pass = document.getElementById('li-pass').value const { data, error } = await supabase.auth.signInWithPassword({ email, password: pass }) if (error) { showErr('li', 'Incorrect email or password.'); return } const meta = data.user.user_metadata enterDash({ first: meta.first, last: meta.last, email, initials: (meta.first[0]+meta.last[0]).toUpperCase() }) }
6
Replace doLogout()
Replace logout
async function doLogout() { await supabase.auth.signOut() showPage('pg-login') toast('Signed out.') }
7
Replace init()
Session persistence (auto-login on refresh)
supabase.auth.getSession().then(({ data }) => { if (data.session) { const meta = data.session.user.user_metadata enterDash({ first: meta.first, last: meta.last, email: data.session.user.email, initials: (meta.first[0]+meta.last[0]).toUpperCase() }) } else { showPage('pg-login') } })
8
Optional • Google OAuth
Enable Google Login
In Supabase Dashboard → Authentication → Providers → Google → enable and paste your Google OAuth credentials. Then update the Google button:
async function signInWithGoogle() { await supabase.auth.signInWithOAuth({ provider: 'google', options: { redirectTo: window.location.origin } }) }
✅
Your app is now production-ready!
All user accounts will be stored securely in Supabase's Postgres database. Passwords are hashed automatically. View all users in your Supabase dashboard under Authentication → Users.
